cat > lambda_function.py <<EOF
import boto3

def lambda_handler(event, context):
    s3 = boto3.client('s3')
    bucket = '<BUCKET_NAME>'
    response = s3.list_objects_v2(Bucket=bucket)
    objects = [obj['Key'] for obj in response.get('Contents', [])]
    print(f"Found objects: {objects}")
    return {
        'statusCode': 200,
        'body': str(objects)
    }
EOF
